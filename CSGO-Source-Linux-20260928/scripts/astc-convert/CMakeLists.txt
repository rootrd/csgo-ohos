cmake_minimum_required(VERSION 3.20)
project(vtf2astc VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT DEFINED ASTCENC_ROOT OR NOT DEFINED KTX_ROOT OR NOT DEFINED JSON_ROOT)
    message(FATAL_ERROR "Use scripts/build-astc-convert.sh to prepare pinned dependencies in Distrobox dev")
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
find_path(KTX_INCLUDE_DIR ktx.h PATHS "${KTX_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(KTX_LIBRARY ktx PATHS "${KTX_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
set(ASTCENC_CLI ON CACHE BOOL "Build the independent reference CLI" FORCE)
set(ASTCENC_ISA_NATIVE ON CACHE BOOL "" FORCE)
set(ASTCENC_WERROR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${ASTCENC_ROOT}" astcenc-build EXCLUDE_FROM_ALL)

add_library(texture_formats src/vpk_reader.cpp src/vtf_decode.cpp src/ktx2_writer.cpp)
target_include_directories(texture_formats PUBLIC src "${KTX_INCLUDE_DIR}" "${JSON_ROOT}")
target_link_libraries(texture_formats PUBLIC "${KTX_LIBRARY}" ZLIB::ZLIB)
target_compile_options(texture_formats PRIVATE -Wall -Wextra -Wpedantic)
add_executable(vtf2astc src/main.cpp)
target_include_directories(vtf2astc PRIVATE "${ASTCENC_ROOT}/Source")
target_link_libraries(vtf2astc PRIVATE texture_formats astcenc-native-static Threads::Threads OpenSSL::Crypto)
target_compile_options(vtf2astc PRIVATE -Wall -Wextra -Wpedantic)

option(VTF2ASTC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(VTF2ASTC_SANITIZE)
    foreach(target texture_formats vtf2astc)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endforeach()
endif()
